# Meson cross-file TEMPLATE to build picolibc for Ouroboros: clang ->
# aarch64-none-elf (compatible with our aarch64-unknown-none), -fPIC so its code
# self-relocates (R_AARCH64_RELATIVE, not the unloadable ABS64), LLVM binutils
# from the Rust toolchain. The @LLVM_BIN@ token is the Rust toolchain's LLVM
# binutils dir; scripts/build-picolibc.sh substitutes it (via `rustc --print
# sysroot`) into a resolved copy before invoking meson, so no host-specific path
# is committed. See libc/ and docs/processes.md's "Writing a program in C".
[binaries]
c = ['clang', '-target', 'aarch64-none-elf', '-nostdlib', '-fPIC']
ar = '@LLVM_BIN@/llvm-ar'
as = '@LLVM_BIN@/llvm-as'
nm = '@LLVM_BIN@/llvm-nm'
strip = '@LLVM_BIN@/llvm-strip'

[host_machine]
system = 'none'
cpu_family = 'aarch64'
cpu = 'aarch64'
endian = 'little'

[properties]
skip_sanity_check = true

# --ld-path is a link option and lives in c_link_args, not on the `c` line
# above. Until 2026-10-16 it rode every compile there: clang warned it unused,
# meson probes each optional flag with -Werror=unused-command-line-argument,
# and 30 of picolibc's 31 optional flags probed NO, so the library was built
# without -frounding-math, -ffunction-sections, -fno-common,
# -fno-stack-protector, and the -fno-builtin, -fno-builtin-malloc and
# -fno-builtin-free picolibc gives the files that need them (without the last,
# nano-malloc's realloc dropped a size store and could hang). Four still probe
# NO, each GCC-only or a link probe -nostdlib cannot pass.
#
# The fixed probes also define eleven _HAVE_ macros in picolibc.h, long
# double and complex among them, and install complex.h. DevTools's cc refuses
# long double here (docs/handoffs/closed/2026-10-15-from-devtools-picolibc-long-double.md),
# so libc/picolibc-patches/0002 undefines the eleven and removes complex.h in
# the installed headers: programs see what they saw before, and the library
# itself is built with them.
[built-in options]
c_link_args = ['--ld-path=@LLVM_BIN@/gcc-ld/ld.lld']
